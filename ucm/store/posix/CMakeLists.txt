file(GLOB_RECURSE UCM_POSIX_STORE_CC_SOURCE_FILES "./cc/*.cc")
add_library(posixstore SHARED ${UCM_POSIX_STORE_CC_SOURCE_FILES})
target_include_directories(posixstore PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/cc)
target_link_libraries(posixstore PUBLIC storeintf infra_logger)

option(ENABLE_GDS "Enable GPUDirect Storage support" OFF)
if(ENABLE_GDS)
    find_library(CUFILE_LIBRARY cufile)
    if(CUFILE_LIBRARY)
        target_link_libraries(posixstore PUBLIC ${CUFILE_LIBRARY})
        target_compile_definitions(posixstore PUBLIC UCM_ENABLE_GDS)
        message(STATUS "GPUDirect Storage support enabled")
    else()
        message(WARNING "cufile library not found, GDS support will be disabled")
    endif()
endif()

file(RELATIVE_PATH INSTALL_REL_PATH ${UCM_ROOT_DIR} ${CMAKE_CURRENT_SOURCE_DIR})
install(TARGETS posixstore LIBRARY DESTINATION ${INSTALL_REL_PATH} COMPONENT ucm)
